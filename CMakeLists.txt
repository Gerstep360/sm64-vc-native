cmake_minimum_required(VERSION 3.20)
project(SM64VCNative LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with Visual Studio -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PLUGIN_SDK "" CACHE PATH "Pinned Plugin-SDK checkout")
set(LIBSM64 "" CACHE PATH "Pinned libsm64 checkout")
set(MINHOOK "" CACHE PATH "Pinned MinHook checkout")
add_subdirectory("${MINHOOK}" "minhook")
if(NOT EXISTS "${PLUGIN_SDK}/shared/Events.h" OR NOT EXISTS "${LIBSM64}/src/libsm64.h")
  message(FATAL_ERROR "Run scripts/build.ps1 first")
endif()
set(sdk_sources)
foreach(name GameVersion PluginBase DynAddress Patch Pattern)
  list(APPEND sdk_sources "${PLUGIN_SDK}/shared/${name}.cpp")
endforeach()
foreach(name CWorld CPools CCamera CPad CTimer CScene CMatrix CEntity CColModel CModelInfo CBaseModelInfo CPhysical CPlaceable CPed CPlayerInfo CPlayerPed CHud)
  list(APPEND sdk_sources "${PLUGIN_SDK}/plugin_vc/game_vc/${name}.cpp")
endforeach()
list(APPEND sdk_sources "${PLUGIN_SDK}/plugin_vc/game_vc/RenderWare.cpp")
list(APPEND sdk_sources "${PLUGIN_SDK}/shared/game/CVector.cpp")
add_library(sm64vc SHARED src/plugin.cpp src/runtime.cpp src/collision.cpp src/render.cpp src/audio.cpp src/hud.cpp src/interaction.cpp src/damage.cpp ${sdk_sources})
target_include_directories(sm64vc PRIVATE src "${PLUGIN_SDK}/shared" "${PLUGIN_SDK}/shared/game"
  "${PLUGIN_SDK}/plugin_vc" "${PLUGIN_SDK}/plugin_vc/game_vc" "${PLUGIN_SDK}/plugin_vc/game_vc/enums"
  "${PLUGIN_SDK}/plugin_vc/game_vc/rw" "${LIBSM64}/src")
target_compile_definitions(sm64vc PRIVATE GTAVC RW PLUGIN_SGV_10EN _CRT_SECURE_NO_WARNINGS
  _CRT_NON_CONFORMING_SWPRINTFS NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(sm64vc PRIVATE /W3 /MP /permissive /Zc:strictStrings- /std:c++latest)
target_link_libraries(sm64vc PRIVATE bcrypt winmm minhook)
target_link_options(sm64vc PRIVATE /Brepro)
set_target_properties(sm64vc PROPERTIES PREFIX "" OUTPUT_NAME "SM64VCNative" SUFFIX ".asi")
